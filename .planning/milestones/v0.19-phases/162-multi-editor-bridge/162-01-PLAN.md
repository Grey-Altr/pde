    ```typescript
    export const RELAY_DEPTH_LIMIT = 1;

    export function validateRelayDepth(req: Request): Response | null {
      const depthHeader = req.headers.get('x-pde-relay-depth');
      if (depthHeader === null) return null;

      const depth = parseInt(depthHeader, 10);
      if (isNaN(depth) || depth < RELAY_DEPTH_LIMIT) return null;

      return new Response(
        JSON.stringify({ error: 'relay_depth_exceeded', received_depth: depth }),
        { status: 400, headers: { 'Content-Type': 'application/json' } }
      );
    }
    ```

    Key rules:
    - Header absent (null) = always allowed (direct client request)
    - depth 0 = allowed (first relay hop, the intended Cursor/Gemini CLI use case)
    - depth >= 1 = rejected with 400 (circular relay attempt)
    - NaN from garbage header = treated as non-relay, allowed (isNaN check returns null)
    - Response body is JSON: `{ error: "relay_depth_exceeded", received_depth: <number> }`
    - Content-Type header set to application/json

    Run tests — all MUST pass.
