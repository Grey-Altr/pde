    **File 1: `dashboard/app/api/mcp/route.ts`**

    Create the Streamable HTTP MCP endpoint. This file is the entry point for all MCP client connections.

    ```typescript
    import { createMcpHandler, withMcpAuth } from 'mcp-handler';
    import { verifyClerkToken } from '@clerk/mcp-tools/next';
    import { auth } from '@clerk/nextjs/server';
    import { registerPdeTools } from '@/lib/mcp/server-factory';
    import { validateOrigin } from '@/lib/mcp/origin-guard';

    export const dynamic = 'force-dynamic';
    export const maxDuration = 300;

    const mcpHandler = createMcpHandler(
      (server) => { registerPdeTools(server); },
      {},
      { basePath: '/api' },
    );

    const authHandler = withMcpAuth(
      mcpHandler,
      async (_, token) => {
        const clerkAuth = await auth({ acceptsToken: 'oauth_token' });
        return verifyClerkToken(clerkAuth, token);
      },
      {
        required: true,
        resourceMetadataPath: '/.well-known/oauth-protected-resource/mcp',
      },
    );

    async function guardedHandler(req: Request) {
      const rejection = validateOrigin(req);
      if (rejection) return rejection;
      return authHandler(req);
    }

    export { guardedHandler as GET, guardedHandler as POST, guardedHandler as DELETE };
    ```

    Key details:
    - `dynamic = 'force-dynamic'` prevents Next.js from caching responses (Pitfall 5 from research)
    - `maxDuration = 300` sets Vercel function timeout to 300s (Hobby max with Fluid Compute)
    - Origin guard runs BEFORE auth handler -- rejects bad origins with 403 before any processing
    - `auth({ acceptsToken: 'oauth_token' })` is critical -- plain `auth()` defaults to session tokens and will NOT recognize MCP OAuth tokens (Pitfall 6)
    - `sessionIdGenerator: undefined` is the default in mcp-handler stateless mode -- do NOT pass it explicitly
    - Export GET, POST, DELETE (MCP spec requires all three for Streamable HTTP)
