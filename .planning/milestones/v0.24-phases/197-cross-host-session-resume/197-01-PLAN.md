SDK cwd sanitization formula (from research, verified against sdk.mjs vJ()):
```javascript
function sanitizeCwdForProjectDir(cwd) {
  const s = cwd.replace(/[^a-zA-Z0-9]/g, '-');
  if (s.length <= 200) return s;
  // SDK uses internal hash for truncation — see Pitfall 1 in RESEARCH.md
  // Use crypto hash for >200 char paths
  const hash = require('node:crypto')
    .createHash('sha256').update(cwd).digest('base64url').slice(0, 8);
  return s.slice(0, 200) + '-' + hash;
}
```
