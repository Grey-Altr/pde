         ```typescript
         const cycleSession = (direction: 'next' | 'prev') => {
           if (sessions.length === 0) return;
           const currentIdx = sessionFilter === 'all' ? -1 : sessionIds.indexOf(sessionFilter);
           if (direction === 'next') {
             const nextIdx = currentIdx + 1 >= sessions.length ? -1 : currentIdx + 1;
             setSessionFilter(nextIdx === -1 ? 'all' : sessionIds[nextIdx]);
           } else {
             const prevIdx = currentIdx <= 0 ? sessions.length - 1 : currentIdx - 1;
             setSessionFilter(currentIdx === -1 ? sessionIds[sessions.length - 1] : sessionIds[prevIdx]);
           }
         };
         ```
