    1. Create dashboard/app/api/cron/gc/route.ts:
    - Import type NextRequest from 'next/server'
    - Import redis from '@/lib/redis'

    - Export async function GET(request: NextRequest): Promise of Response
    - Auth guard: read request.headers.get('authorization'), check process.env.CRON_SECRET exists AND authHeader equals template-string Bearer followed by CRON_SECRET. Return new Response('Unauthorized', { status: 401 }) if check fails
    - Compute cutoffMs = Date.now() - 7 * 24 * 60 * 60 * 1000
    - Call redis.zrange('pde:default:sessions', '-inf', cutoffMs, { byScore: true }) to get stale session IDs
    - If staleIds is non-empty array:
      - Create pipeline: const p = redis.pipeline()
      - For each id in staleIds (cast as string[]):
        - p.del(template-string pde:default:events:{id})
        - p.del(template-string pde:default:session:{id})
        - p.zrem('pde:default:sessions', id)
      - await p.exec()
      - Set deleted.count = staleIds.length
    - Return Response.json({ ok: true, deleted: { count }, cutoff: new Date(cutoffMs).toISOString(), ran_at: new Date().toISOString() })
