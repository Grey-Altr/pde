Add a `case 'image':` block to the main switch statement in pde-tools.cjs, following the exact same pattern as `case 'cli-anything':`. The image case routes to 6 subcommands:

```javascript
case 'image': {
  const subcommand = args[1];
  if (subcommand === 'og') {
    const { generateOgImage } = require('./lib/image-pipeline/og.cjs');
    // Parse args: --title, --description, --slug (required), --template (optional)
    const title = getFlag(args, '--title') || 'Untitled';
    const description = getFlag(args, '--description') || '';
    const slug = getFlag(args, '--slug') || 'og-image';
    const result = await generateOgImage({ title, description, slug });
    console.log(JSON.stringify(result.meta, null, 2));
  } else if (subcommand === 'social') {
    const { generateSocialCards } = require('./lib/image-pipeline/social.cjs');
    const title = getFlag(args, '--title') || 'Untitled';
    const description = getFlag(args, '--description') || '';
    const slug = getFlag(args, '--slug') || 'social-card';
    const results = await generateSocialCards({ title, description, slug });
    console.log(JSON.stringify(results.map(r => r.meta), null, 2));
  } else if (subcommand === 'screenshot') {
    const { captureScreenshot } = require('./lib/image-pipeline/screenshot.cjs');
    const url = args[2]; // positional
    const viewport = getFlag(args, '--viewport') || 'desktop';
    const slug = getFlag(args, '--slug') || 'screenshot';
    const format = getFlag(args, '--format') || 'png';
    const result = await captureScreenshot({ url, viewport, slug, format });
    console.log(JSON.stringify(result.meta, null, 2));
  } else if (subcommand === 'mockup') {
    const { generateMockup } = require('./lib/image-pipeline/mockup.cjs');
    const screenshotPath = args[2]; // positional
    const frame = getFlag(args, '--frame') || 'browser';
    const slug = getFlag(args, '--slug') || 'mockup';
    const result = await generateMockup({ screenshotPath, frame, slug });
    console.log(JSON.stringify(result.meta, null, 2));
  } else if (subcommand === 'rembg') {
    const { removeBackground } = require('./lib/image-pipeline/rembg.cjs');
    const inputPath = args[2]; // positional
    const slug = getFlag(args, '--slug') || 'rembg';
    const result = await removeBackground({ inputPath, slug });
    if (result) {
      console.log(JSON.stringify(result.meta, null, 2));
    } else {
      console.log(JSON.stringify({ status: 'skipped', reason: 'no API key' }));
    }
  } else if (subcommand === 'list') {
    const { listAssets } = require('./lib/image-pipeline/assets.cjs');
    const type = getFlag(args, '--type') || undefined;
    const assets = listAssets({ type });
    console.log(JSON.stringify(assets, null, 2));
  } else {
    console.error(`Unknown image subcommand: ${subcommand}. Available: og, social, screenshot, mockup, rembg, list`);
    process.exit(1);
  }
  break;
}
```

Create `commands/image.md` following the established command documentation pattern from existing commands/ files. The document should describe:

1. **Command name:** `/pde:image`
2. **Description:** Generate production-ready images — OG images, social cards, device mockups, screenshots, and background removal.
3. **Subcommands:**
   - `og` — Generate OG image (1200x630) from product data using Satori. Options: `--title`, `--description`, `--slug`, `--template`.
   - `social` — Generate social media cards (Twitter 1200x628, LinkedIn 1200x627, Facebook 1200x630) from one data input. Options: `--title`, `--description`, `--slug`.
   - `screenshot` — Capture product screenshot via Playwright. Positional: URL. Options: `--viewport` (desktop|tablet|mobile|WxH), `--slug`, `--format` (png|jpg), `--timeout`.
   - `mockup` — Composite screenshot onto device frame. Positional: screenshot path. Options: `--frame` (browser|phone), `--slug`.
   - `rembg` — Remove image background via remove.bg API. Positional: image path. Options: `--slug`. Requires `REMOVEBG_API_KEY` env var. Graceful skip without it. 50/month free tier limit.
   - `list` — List all generated assets with metadata. Options: `--type` (og|social|mockup|screenshot|rembg).
4. **Asset storage:** All outputs go to `.planning/design/assets/{type}/{slug}-{timestamp}.png` with JSON metadata sidecar.
5. **Prerequisites:** `npm install satori @resvg/resvg-js sharp playwright htm && npx playwright install chromium`
6. **Examples:** One example per subcommand showing typical usage.

Follow the exact markdown structure used by other commands in the commands/ directory. Include the execution pattern that references `node bin/pde-tools.cjs image <subcommand>`.
