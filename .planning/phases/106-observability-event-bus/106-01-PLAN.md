**1. Create `bin/pane-experiment.sh`** following the established pane script pattern exactly:

```bash
#!/usr/bin/env bash
# pane-experiment.sh -- streams experiment lifecycle events
# OBS-02: Shows current iteration, best metric, keep/discard ratio, budget remaining

NDJSON="${1:-}"
if [ -z "$NDJSON" ]; then
  echo "Usage: pane-experiment.sh <ndjson-path>"
  exit 1
fi

echo "[ experiment ] waiting for events..."
echo ""

# State tracking variables
ITERATION=0
BEST_METRIC="--"
KEEPS=0
DISCARDS=0
CRASHES=0
BUDGET=0
SLUG=""
STATUS="idle"

tail -F "${NDJSON}" 2>/dev/null | while IFS= read -r line; do
  event_type=$(echo "$line" | jq -r '.event_type // empty' 2>/dev/null)

  case "$event_type" in
    experiment.start)
      SLUG=$(echo "$line" | jq -r '.slug // ""' 2>/dev/null)
      BUDGET=$(echo "$line" | jq -r '.iteration_budget // 0' 2>/dev/null)
      BEST_METRIC=$(echo "$line" | jq -r '.baseline_metric // "--"' 2>/dev/null)
      ITERATION=0
      KEEPS=0
      DISCARDS=0
      CRASHES=0
      STATUS="running"
      ;;
    experiment.iteration)
      ITERATION=$(echo "$line" | jq -r '.iteration // 0' 2>/dev/null)
      ;;
    experiment.keep)
      KEEPS=$(( KEEPS + 1 ))
      val=$(echo "$line" | jq -r '.metric_value // empty' 2>/dev/null)
      [ -n "$val" ] && BEST_METRIC="$val"
      ;;
    experiment.discard)
      DISCARDS=$(( DISCARDS + 1 ))
      ;;
    experiment.crash)
      CRASHES=$(( CRASHES + 1 ))
      ;;
    experiment.complete)
      BEST_METRIC=$(echo "$line" | jq -r '.best_metric // "--"' 2>/dev/null)
      STATUS="complete"
      ;;
    *) continue ;;
  esac

  # Redraw display (cursor to line 3, clear downward)
  printf '\033[3;1H\033[J'
  if [ "$STATUS" = "idle" ]; then
    printf '  No active experiment\n'
  else
    printf '  Experiment:    %s\n' "$SLUG"
    printf '  Iteration:     %s / %s\n' "$ITERATION" "$BUDGET"
    printf '  Best metric:   %s\n' "$BEST_METRIC"
    printf '  Keep/Discard:  %s / %s\n' "$KEEPS" "$DISCARDS"
    printf '  Crashes:       %s\n' "$CRASHES"
    REMAINING=$(( BUDGET - ITERATION ))
    [ "$REMAINING" -lt 0 ] && REMAINING=0
    printf '  Budget left:   %s iterations\n' "$REMAINING"
    if [ "$STATUS" = "complete" ]; then
      printf '\n  \033[1;32mExperiment complete\033[0m\n'
    fi
  fi
done
```
